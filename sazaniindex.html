const MASTER_SHEET_ID = '1GjuKHV4JRBBUhzNPlrgLa2exUPcqAXIbxXaVKs6re9o'; const CALENDAR_TAB = 'Room Calendar'; const OUTPUT_TAB = 'Availability'; const DATE_START = new Date('2026-01-01'); const DATE_END = new Date('2029-12-31'); const ROOM_NAMES = { 'Hiraeth House Two': 'Hiraeth 2', 'Hiraeth House Three': 'Hiraeth 3', 'Hiraeth House Four': 'Hiraeth 4', 'Kas Kazi': 'Kas Kazi', 'Kusi': 'Kusi', 'Banda': 'Banda', }; function formatDate(val) { var d = new Date(val); var y = d.getUTCFullYear(); var m = String(d.getUTCMonth() + 1).padStart(2, '0'); var day = String(d.getUTCDate()).padStart(2, '0'); return y + '-' + m + '-' + day; } function addOneDay(val) { var d = new Date(val); d.setUTCDate(d.getUTCDate() + 1); return formatDate(d); } function extendCalendar(calendar) { // Read existing data to find current date range var data = calendar.getDataRange().getValues(); var dateRow = 4; var headers = data[dateRow]; // Find the last column that has a date var lastDateCol = 0; for (var c = 1; c < headers.length; c++) { if (headers[c] && typeof headers[c] !== 'string') lastDateCol = c; } // Find the last date currently in the sheet var lastDate = lastDateCol > 0 ? new Date(headers[lastDateCol]) : new Date(DATE_START); lastDate.setUTCDate(lastDate.getUTCDate() + 1); // start from day after last // Build list of missing dates up to DATE_END var missingDates = []; var d = new Date(lastDate); while (d <= DATE_END) { missingDates.push(new Date(d)); d.setUTCDate(d.getUTCDate() + 1); } if (missingDates.length === 0) { Logger.log('Calendar already covers full date range — no extension needed.'); return; } // Write missing date headers starting from next available column var startCol = lastDateCol + 1; var dateValues = [missingDates.map(function(dt) { return dt; })]; calendar.getRange(dateRow + 1, startCol + 1, 1, missingDates.length).setValues(dateValues); // Format the new date header cells as dates calendar.getRange(dateRow + 1, startCol + 1, 1, missingDates.length) .setNumberFormat('d mmm yy'); Logger.log('Extended calendar by ' + missingDates.length + ' days to ' + formatDate(DATE_END)); } function syncAvailability() { var master = SpreadsheetApp.openById(MASTER_SHEET_ID); var calendar = master.getSheetByName(CALENDAR_TAB); if (!calendar) { Logger.log('ERROR: Could not find "' + CALENDAR_TAB + '" in master sheet.'); return; } // Extend the calendar date range first extendCalendar(calendar); // Re-read data after extension var data = calendar.getDataRange().getValues(); var headerRow = 4; var roomStart = 5; var headers = data[headerRow]; var rows = [['Checkin', 'Checkout', 'Room', 'Status']]; for (var r = roomStart; r < data.length; r++) { var rawRoom = String(data[r][0]).trim(); if (!rawRoom) continue; var roomName = ROOM_NAMES[rawRoom] || rawRoom; var currentInitials = null; var currentStart = null; var currentLast = null; for (var c = 1; c < headers.length; c++) { var header = headers[c]; if (!header || typeof header === 'string') continue; var cell = String(data[r][c]).trim(); if (cell && cell !== 'AV') { if (cell === currentInitials) { currentLast = header; } else { if (currentInitials) { rows.push([formatDate(currentStart), addOneDay(currentLast), roomName, 'Booked']); } currentInitials = cell; currentStart = header; currentLast = header; } } else { if (currentInitials) { rows.push([formatDate(currentStart), addOneDay(currentLast), roomName, 'Booked']); currentInitials = null; currentStart = null; currentLast = null; } } } if (currentInitials) { rows.push([formatDate(currentStart), addOneDay(currentLast), roomName, 'Booked']); } } // Sort chronologically by checkin var headerRow2 = rows.shift(); rows.sort(function(a, b) { return new Date(a[0]) - new Date(b[0]); }); rows.unshift(headerRow2); // Write to public Availability sheet var thisSheet = SpreadsheetApp.getActiveSpreadsheet(); var output = thisSheet.getSheetByName(OUTPUT_TAB); if (!output) output = thisSheet.insertSheet(OUTPUT_TAB); output.clearContents(); output.getRange(1, 1, rows.length, 4).setValues(rows); output.setFrozenRows(1); output.getRange(1, 1, 1, 4).setFontWeight('bold'); Logger.log('Done. ' + (rows.length - 1) + ' bookings written.'); } function createHourlyTrigger() { var triggers = ScriptApp.getProjectTriggers(); for (var i = 0; i < triggers.length; i++) { if (triggers[i].getHandlerFunction() === 'syncAvailability') { ScriptApp.deleteTrigger(triggers[i]); } } ScriptApp.newTrigger('syncAvailability') .timeBased() .everyHours(1) .create(); Logger.log('Hourly trigger created.'); } function debugSheetNames() { var master = SpreadsheetApp.openById(MASTER_SHEET_ID); var sheets = master.getSheets(); for (var i = 0; i < sheets.length; i++) { Logger.log('Tab ' + i + ': "' + sheets[i].getName() + '"'); } }
